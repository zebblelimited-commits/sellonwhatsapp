How to send it

From your server-side order creation/payment flow:


import { sendEmail } from "@/lib/email";
import OrderConfirmation from "@/emails/OrderConfirmation";

await sendEmail({
  to: {
    email: order.customerEmail,
    name: order.customerName,
  },

  subject: `Order #${order.orderNumber} received 🎉`,

  react: (
    <OrderConfirmation
      customerName={order.customerName}
      orderNumber={order.orderNumber}
      storeName={order.storeName}
      items={order.items}
      subtotal={order.subtotal}
      shipping={order.shipping}
      discount={order.discount}
      total={order.total}
      orderUrl={`https://sellonwhatsapp.com/orders/${order.id}`}
    />
  ),

  type: "order_confirmation",

  idempotencyKey: `order_confirmation:${order.id}`,

  metadata: {
    orderId: order.id,
    customerId: order.customerId,
    storeId: order.storeId,
  },
});

Use this after your payment webhook or server-side payment verification confirms the transaction:

This should happen after payment has been confirmed, not merely when a customer starts checkout.

import { sendEmail } from "@/lib/email";

await sendEmail({

Example: sending a shipping update

When your courier integration changes the order status:

import { sendEmail } from "@/lib/email";
import ShippingUpdate from "@/emails/ShippingUpdate";

await sendEmail({
  to: {
    email: order.customerEmail,
    name: order.customerName,
  },

  subject: `Shipping update for order #${order.orderNumber} 📦`,

  react: (
    <ShippingUpdate
      customerName={order.customerName}
      orderNumber={order.orderNumber}
      storeName={order.storeName}
      courierName={shipment.courierName}
      trackingNumber={shipment.trackingNumber}
      trackingUrl={shipment.trackingUrl}
      status="Shipped"
      statusDescription="Your order has been handed over to the courier and is now on its way."
      estimatedDelivery={shipment.estimatedDelivery}
      steps={[
        {
          label: "Order placed",
          description: "Your order was received.",
          completed: true,
          date: order.createdAt,
        },
        {
          label: "Payment confirmed",
          description: "Your payment was successfully confirmed.",
          completed: true,
          date: payment.confirmedAt,
        },
        {
          label: "Order shipped",
          description: "Your order has been handed over to the courier.",
          completed: true,
          current: true,
          date: shipment.shippedAt,
        },
        {
          label: "Out for delivery",
          description: "Your order will be delivered to you.",
          completed: false,
        },
        {
          label: "Delivered",
          description: "Your order has been delivered.",
          completed: false,
        },
      ]}
      orderUrl={`https://sellonwhatsapp.com/orders/${order.id}`}
    />
  ),

  type: "shipping_update",

  idempotencyKey: `shipping_update:${order.id}:${shipment.status}`,

  metadata: {
    orderId: order.id,
    customerId: order.customerId,
    storeId: order.storeId,
    courierName: shipment.courierName,
    trackingNumber: shipment.trackingNumber,
    shippingStatus: shipment.status,
  },
});
